Restriccions de claus foranes

Taula Clau Forana Nuls Esborrat Modificació Justificació
EMPLEAT id_persona No Restringit Restringit Una persona no pot ser eliminada si és un empleat actiu. No es pot modificar la relació perquè un empleat no pot canviar d'identitat.
CLIENT id_persona No Restringit Restringit Una persona no pot ser eliminada si té reserves associades com a client. La seva informació no pot ser canviada si ja ha fet reserves.
RESERVA id_client No Restringit Restringit Un client no es pot eliminar si té reserves actives. Tampoc es pot modificar perquè les reserves han de mantenir el client original.
RESERVA id_habitacio No Restringit Restringit Una habitació no es pot eliminar si té reserves associades. No es pot modificar perquè la reserva ha de mantenir l'habitació assignada.
FACTURA id_reserva No Restringit Restringit Una reserva no es pot eliminar si té una factura associada. Les factures han de mantenir la reserva original sense modificacions.
REALITZAR id_empleat No Propagar Restringit Si un empleat s'elimina, les tasques realitzades han de ser eliminades també. No es poden modificar perquè han de reflectir l'empleat original.
REALITZAR id_tasca No Propagar Restringit Si una tasca s'elimina, les seves assignacions desapareixen. No es poden modificar per mantenir la coherència de l'historial de treball.

Creació de taules

CREATE TABLE persona (
    id_persona INT AUTO_INCREMENT PRIMARY KEY,
    nom VARCHAR(255) NOT NULL,
    cognom VARCHAR(255) NOT NULL,
    adreca VARCHAR(255) NOT NULL,
    dni VARCHAR(20) UNIQUE NOT NULL,
    data_naixement DATE NOT NULL,
    telefon VARCHAR(20) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL
);

CREATE TABLE empleat (
    id_empleat INT PRIMARY KEY,
    lloc_feina VARCHAR(255) NOT NULL,
    data_contractacio DATE NOT NULL,
    salari_brut DOUBLE NOT NULL,
    estat_laboral VARCHAR(50) NOT NULL,
    FOREIGN KEY (id_empleat) REFERENCES PERSONA(id_persona)
        ON DELETE RESTRICT
        ON UPDATE RESTRICT
);

CREATE TABLE tasca (
    id_tasca INT AUTO_INCREMENT PRIMARY KEY,
    data_execucio DATE NOT NULL,
    data_creacio DATE NOT NULL,
    descripcio TEXT UNIQUE NOT NULL,
    estat VARCHAR(16) NOT NULL
);

CREATE TABLE client (
    id_client INT PRIMARY KEY,
    data_registre DATE NOT NULL,
    tipus_client VARCHAR(50) NOT NULL,
    targeta_credit VARCHAR(50) NOT NULL,
    FOREIGN KEY (id_client) REFERENCES PERSONA(id_persona)
        ON DELETE RESTRICT
        ON UPDATE RESTRICT
);

CREATE TABLE habitacio (
    id_habitacio INT AUTO_INCREMENT PRIMARY KEY,
    numero_habitacio INT UNIQUE NOT NULL,
    tipus VARCHAR(50) NOT NULL,
    capacitat INT NOT NULL,
    preu_nit_AD DOUBLE NOT NULL,
    preu_nit_MP DOUBLE NOT NULL,
    estat VARCHAR(50) NOT NULL,
    descripcio TEXT
);

CREATE TABLE reserva (
    id_reserva INT AUTO_INCREMENT PRIMARY KEY,
    data_reserva DATE NOT NULL,
    data_inici DATE NOT NULL,
    data_fi DATE NOT NULL,
    tipus_reserva VARCHAR(50) NOT NULL,
    tipus_IVA VARCHAR(15) NOT NULL,
    preu_total_reserva DOUBLE NOT NULL,
    id_client INT NOT NULL,
    id_habitacio INT NOT NULL,
    FOREIGN KEY (id_client) REFERENCES CLIENT(id_client)
        ON DELETE RESTRICT
        ON UPDATE RESTRICT,
    FOREIGN KEY (id_habitacio) REFERENCES HABITACIO(id_habitacio)
        ON DELETE RESTRICT
        ON UPDATE RESTRICT
);

CREATE TABLE factura (
    id_factura INT AUTO_INCREMENT PRIMARY KEY,
    data_emissio DATE NOT NULL,
    metode_pagament VARCHAR(50) NOT NULL,
    base_imposable DOUBLE NOT NULL,
    tipus_IVA VARCHAR(16) NOT NULL,
    total DOUBLE NOT NULL,
    id_reserva INT NOT NULL,
    FOREIGN KEY (id_reserva) REFERENCES RESERVA(id_reserva)
        ON DELETE RESTRICT
        ON UPDATE RESTRICT
);

CREATE TABLE realitzar (
    id_empleat INT NOT NULL,
    id_tasca INT NOT NULL,
    PRIMARY KEY (id_empleat, id_tasca),
    FOREIGN KEY (id_empleat) REFERENCES EMPLEAT(id_empleat)
        ON DELETE CASCADE
        ON UPDATE RESTRICT,
    FOREIGN KEY (id_tasca) REFERENCES TASCA(id_tasca)
        ON DELETE CASCADE
        ON UPDATE RESTRICT
);
